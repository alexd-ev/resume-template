%-------------------------------------------------------------------------------
% Resume layout helpers
%-------------------------------------------------------------------------------
% A full-width rule separating the header and each resume section.
\newcommand{\resumerule}{%
    \par\noindent\hspace*{3pt}%
    \textcolor{rulegray}{\rule{\dimexpr\textwidth-3pt\relax}{1pt}}\par
}

% A resume section: a rule, then a bold underlined heading.
\newcommand{\resumesection}[1]{%
    \par\vspace{-1.7pt}%
    \resumerule
    \vspace{0.5pt}%
    \noindent\textbf{\underline{#1}}\par
}

\newlist{resumelist}{itemize}{1}
\setlist[resumelist]{
    label=$\bullet$,
    leftmargin=36pt,
    labelwidth=10pt,
    labelsep=13pt,
    topsep=0pt,
    itemsep=0pt,
    parsep=0pt,
    partopsep=0pt,
}
